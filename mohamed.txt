# 💾 تصدير نسخة من المشروع والمحادثة

---

## 🗂️ أولاً: تصدير المحادثة كاملة

### من ChatGPT / المتصفح:

```
1. اضغط على صورتك (أعلى اليمين)
2. Settings ← إعدادات
3. Data controls ← التحكم في البيانات
4. Export data ← تصدير البيانات
5. Confirm export
6. ✅ سيصلك إيميل فيه ملف ZIP بكل محادثاتك
```

### الطريقة اليدوية السريعة:

```
1. اضغط Ctrl + A  (تحديد الكل)
2. اضغط Ctrl + C  (نسخ)
3. افتح Word أو Notepad
4. اضغط Ctrl + V  (لصق)
5. احفظ باسم: matjary-project.docx
```

### طباعة كـ PDF:

```
1. اضغط Ctrl + P
2. اختر "Save as PDF"
3. احفظ باسم: matjary-chat.pdf
```

## ⚡ ثانياً: سكريبت ينشئ المشروع تلقائياً

### 🍎 على Mac / Linux — ملف `setup.sh`

```bash
#!/bin/bash
# ═══════════════════════════════════════
#  متجري — سكريبت إنشاء المشروع
# ═══════════════════════════════════════

echo "🛍️  جاري إنشاء مشروع متجري..."

mkdir -p matjary-store/src/{lib,stores,data,types,components/{ui,store},pages}
cd matjary-store

# ── package.json ──
cat > package.json << 'EOF'
{
  "name": "matjary-store",
  "private": true,
  "version": "1.0.0",
  "type": "module",
  "scripts": {
    "dev": "vite",
    "build": "tsc -b && vite build",
    "preview": "vite preview"
  },
  "dependencies": {
    "lucide-react": "^0.575.0",
    "react": "^19.2.0",
    "react-dom": "^19.2.0",
    "react-router-dom": "^7.6.0",
    "zustand": "^5.0.15"
  },
  "devDependencies": {
    "@tailwindcss/vite": "^4.2.1",
    "@types/react": "^19.2.0",
    "@types/react-dom": "^19.2.0",
    "@vitejs/plugin-react": "^5.2.0",
    "tailwindcss": "^4.2.1",
    "typescript": "^5.8.3",
    "vite": "^6.3.0"
  }
}
EOF

# ── vite.config.ts ──
cat > vite.config.ts << 'EOF'
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';
import tailwindcss from '@tailwindcss/vite';
import path from 'path';

export default defineConfig({
  plugins: [react(), tailwindcss()],
  resolve: { alias: { '@': path.resolve(__dirname, './src') } },
});
EOF

# ── tsconfig.json ──
cat > tsconfig.json << 'EOF'
{
  "compilerOptions": {
    "target": "ES2020",
    "lib": ["ES2020", "DOM", "DOM.Iterable"],
    "module": "ESNext",
    "skipLibCheck": true,
    "moduleResolution": "bundler",
    "allowImportingTsExtensions": true,
    "isolatedModules": true,
    "moduleDetection": "force",
    "noEmit": true,
    "jsx": "react-jsx",
    "strict": true,
    "noUnusedLocals": false,
    "noUnusedParameters": false,
    "baseUrl": ".",
    "paths": { "@/*": ["./src/*"] }
  },
  "include": ["src"]
}
EOF

# ── index.html ──
cat > index.html << 'EOF'
<!DOCTYPE html>
<html lang="ar" dir="rtl">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>متجري | Matjary</title>
    <link rel="preconnect" href="https://fonts.googleapis.com" />
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
    <link href="https://fonts.googleapis.com/css2?family=DM+Sans:wght@300;400;500;600;700&family=Noto+Sans+Arabic:wght@300;400;500;600;700&display=swap" rel="stylesheet" />
  </head>
  <body>
    <div id="root"></div>
    <script type="module" src="/src/main.tsx"></script>
  </body>
</html>
EOF

# ── .gitignore ──
cat > .gitignore << 'EOF'
node_modules
dist
.DS_Store
*.local
EOF

# ── README.md ──
cat > README.md << 'EOF'
# 🛍️ متجري — Matjary

متجر إلكتروني عربي/إنجليزي لملابس النساء والأطفال.

## التشغيل
npm install
npm run dev

## النشر
Build: npm run build
Output: dist

## لوحة الإدارة
/admin-login
EOF

# ── src/main.tsx ──
cat > src/main.tsx << 'EOF'
import { StrictMode } from 'react';
import { createRoot } from 'react-dom/client';
import './index.css';
import App from './App';

createRoot(document.getElementById('root')!).render(
  <StrictMode><App /></StrictMode>
);
EOF

echo ""
echo "✅ تم إنشاء الهيكل الأساسي!"
echo ""
echo "📋 المتبقي: الصق محتوى هذه الملفات من المحادثة:"
echo "   • src/index.css"
echo "   • src/App.tsx"
echo "   • src/types.ts"
echo "   • src/data/products.ts"
echo "   • src/stores/shop.ts"
echo "   • src/components/store/StoreComponents.tsx"
echo "   • src/components/ProtectedRoute.tsx"
echo "   • src/pages/Home.tsx"
echo "   • src/pages/Collection.tsx"
echo "   • src/pages/ProductDetail.tsx"
echo "   • src/pages/Checkout.tsx"
echo "   • src/pages/Admin.tsx"
echo "   • src/pages/AdminLogin.tsx"
echo ""
echo "🚀 ثم شغّل:  cd matjary-store && npm install && npm run dev"
```

**طريقة التشغيل:**
```bash
chmod +x setup.sh
./setup.sh
```
